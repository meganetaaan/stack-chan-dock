cmake_minimum_required(VERSION 3.22)
project(irte_pocket CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_NFKC_COMPILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../pocket-eval/upstream/sentencepiece-31646a467d2051eb904e0b45de3a73e91fe1c1e3" sentencepiece EXCLUDE_FROM_ALL)
add_library(irte_pocket SHARED ../pocket-tokenizer.cpp)
target_include_directories(irte_pocket PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../pocket-eval/upstream/sentencepiece-31646a467d2051eb904e0b45de3a73e91fe1c1e3/src")
target_link_libraries(irte_pocket sentencepiece-static)
if(ANDROID)
  target_link_libraries(irte_pocket log)
else()
  target_include_directories(irte_pocket PRIVATE "${JAVA_INCLUDE_DIR}" "${JAVA_INCLUDE_DIR}/win32")
  target_link_options(irte_pocket PRIVATE -static-libgcc -static-libstdc++)
endif()
target_compile_options(irte_pocket PRIVATE -O3 -fvisibility=hidden)
if(ANDROID)
  target_link_options(irte_pocket PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
endif()
